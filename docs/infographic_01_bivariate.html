ตรวจว่าเกณฑ์เรียงน้อยไปมาก จำนวน = n−1 · จัดชั้นด้วย np.digitize — ค่าเท่าเกณฑ์ขึ้นชั้นบน (300 พอดี → ชั้นกลาง) · ไม่มีข้อมูล = −1

np.digitize
def classify(self, metric_vals, valid):
    c = np.digitize(metric_vals, self.breaks).astype('int16')
    c[~valid] = -1          # ไม่มีข้อมูล
    return c
# breaks = [300, 1500]
#   120 → 0 ,  300 → 1 ,  999 → 1 ,  1500 → 2
